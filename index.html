function parseNum(v) {

  if (typeof v === "number") return v;

  var s = String(v || "").trim().replace(/[^\d.,-]/g, "");

  if (s.indexOf(".") !== -1 && s.indexOf(",") !== -1) s = s.replace(/\./g, "").replace(",", ".");

  else if (s.indexOf(",") !== -1) s = s.replace(",", ".");

  return parseFloat(s) || 0;

}

  ates.sort(function (a, b) { return a - b; });

  for (var i = 0; i < ates.length; i++) {

    if (preco <= ates[i]) return obj["ate" + ates[i]];

  }

  return obj.acima79 != null ? obj.acima79 : obj.acima50 != null ? obj.acima50 : obj.acima200 != null ? obj.acima200 : obj.acima10 != null ? obj.acima10 : 0;

}

function indicePeso(pesoKg) {

  for (var i = 0; i < ENVIO_ML.faixasPeso.length; i++) {

    if (pesoKg <= ENVIO_ML.faixasPeso[i].pesoMax) return i;

  }

  return ENVIO_ML.faixasPeso.length - 1;

}

function custoEnvioML(pesoKg, precoProduto) {

  var linha = ENVIO_ML.faixasPeso[indicePeso(pesoKg)];

  var coluna = -1;

  for (var i = 0; i < ENVIO_ML.faixasPreco.length; i++) {

    if (precoProduto <= ENVIO_ML.faixasPreco[i].limite) { coluna = i; break; }

  }

  if (coluna === -1) coluna = ENVIO_ML.faixasPreco.length - 1;

  return linha.custos[coluna];

}

function maiorLimite(obj) {

  var m = 0;

  Object.keys(obj).forEach(function (k) {

    if (k.indexOf("ate") === 0) m = Math.max(m, parseFloat(k.replace("ate", "")));

  });

  return m || 79;

}

function calcularPreco(entradas) {

  var custo = entradas.custo || 0;

  var cnpj = entradas.cnpj || "";

  var nivel = entradas.nivel || 5;

  var pesoKg = entradas.pesoKg || 0;

  var plano = entradas.plano || "classico";

  var valorManual = entradas.valorManual || 0;

  var valorLiq = entradas.valorLiq || 0;

  var pctLiq = entradas.pctLiq || 0;


  if (CONFIG.usarInsumos) {

    custo += (pesoKg <= 0.3 ? CONFIG.insumos.ate300G : CONFIG.insumos.acima300G);

  }


  var aliquota = CONFIG.aliq[cnpj] != null ? CONFIG.aliq[cnpj] : 0.10;

  var idxPeso = indicePeso(pesoKg);

  var resultados = {};


  Object.keys(CONFIG.canais).forEach(function (canal) {

    var cfg = CONFIG.canais[canal];

    var comissaoPct;

    if (typeof cfg.comissao === "number") comissaoPct = cfg.comissao;

    else if (cfg.comissao[plano] !== undefined) comissaoPct = cfg.comissao[plano];

    else comissaoPct = porFaixa(cfg.comissao, 0);


    var denominador = 1 - (aliquota + comissaoPct);

    var constNivel = cfg.niveis[nivel] != null ? cfg.niveis[nivel] : 1;


    function fretePara(preco) {

      if (canal === "mercadoLivre") return custoEnvioML(pesoKg, preco);

      if (typeof cfg.frete === "number") return cfg.frete;

      if (canal === "presencial") return 0;

      if (canal === "rd") return (cfg.freteACIMA[idxPeso] || 0) * cfg.comissao;

      if (cfg.freteATE && preco <= maiorLimite(cfg.freteATE)) return porFaixa(cfg.freteATE, preco) * constNivel;

      return (cfg.freteACIMA[idxPeso] || 0) * constNivel;

    }

    function taxaPara(preco) {

      if (!cfg.taxa) return 0;

      return porFaixa(cfg.taxa, preco);

    }


    var manual = null;

    if (valorManual > 0) {

      manual = valorManual - (custo + valorManual * (aliquota + comissaoPct) + fretePara(valorManual) + taxaPara(valorManual));

    }

    var vliq = null;

    if (valorLiq > 0) {

      vliq = (valorLiq + custo + fretePara(valorLiq) + taxaPara(valorLiq)) / denominador;

    }

    var pct = null;

    if (pctLiq > 0) {

      var custoComMargem = custo * (1 + pctLiq / 100);

      pct = (custoComMargem + fretePara(custoComMargem) + taxaPara(custoComMargem)) / denominador;

    }

    resultados[canal] = { manual: manual, vliq: vliq, pct: pct };

  });


  return resultados;

}

  });

}
